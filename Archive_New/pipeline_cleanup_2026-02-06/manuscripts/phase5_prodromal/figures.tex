% Figure 2: Cox Model Analysis (Survival Curves)
\begin{figure*}[t]
\centering
\includegraphics[width=0.95\textwidth]{figures/cox_model_analysis.png}
\caption{\textbf{DeepSurv Outperforms Cox Proportional Hazards.}
\textbf{(A)} Time-dependent ROC curves at 2, 5, 10 years. DeepSurv (solid) vs. Cox (dashed). 5-year AUC: 0.78 vs. 0.71 ($p<0.05$).
\textbf{(B)} Concordance index (C-index) with 95\% CI: DeepSurv 0.76 (0.71-0.81) vs. Cox 0.68 (0.62-0.74), $p=0.003$.
\textbf{(C)} Calibration curves: Predicted vs. observed 5-year survival. DeepSurv well-calibrated (Hosmer-Lemeshow $p=0.28$), Cox slight miscalibration.
\textbf{(D)} Brier score over time: DeepSurv consistently lower (better) than Cox (0.14 vs. 0.19 at 5 years).
}
\label{fig:performance}
\end{figure*}

% Figure 3: DeepSurv Neural Survival Analysis
\begin{figure*}[t]
\centering
\includegraphics[width=0.95\textwidth]{figures/deepsurv_analysis.png}
\caption{\textbf{Cox Model Hazard Ratios for Conversion Risk Factors.}
\textbf{(A)} Forest plot of multivariable Cox model. Baseline UPDRS-III dominant (HR 3.8 per 5 points, 95\% CI 2.3-6.2), followed by RBD (3.2, 1.8-5.7), GBA (3.5, 1.9-6.4), DaTscan SBR (2.8 per 0.5 decrease, 1.7-4.6), hyposmia (2.4, 1.4-4.1).
\textbf{(B)} Kaplan-Meier curves stratified by baseline UPDRS-III tertiles: <8 (low), 8-12 (medium), >12 (high). High tertile median time-to-conversion 2.8 years vs. not reached for low (log-rank $p<0.001$).
\textbf{(C)} DeepSurv feature importance (permutation-based): Baseline UPDRS-III (0.38), DaTscan SBR (0.24), RBD (0.18), GBA (0.12), hyposmia (0.09).
}

\end{figure*}

% Figure 4: Risk Stratification Dashboard
\begin{figure*}[t]
\centering
\includegraphics[width=0.95\textwidth]{figures/risk_stratification_dashboard.png}
\caption{\textbf{Risk Stratification Identifies High-Conversion Stratum.}
\textbf{(A)} Kaplan-Meier curves by DeepSurv-predicted risk tertiles: Low (<33rd percentile), Medium (33-67th), High (>67th). 5-year conversion: Low 4\%, Medium 15\%, High 31\% (log-rank $p<0.001$).
\textbf{(B)} Hazard ratios: High vs. Low HR 12.4 (95\% CI 6.1-25.2), High vs. Medium HR 2.8 (1.6-4.9).
\textbf{(C)} ROC curve for high-risk classification (top tertile). AUC 0.84, sensitivity 89\%, specificity 78\% at optimal threshold.
\textbf{(D)} High-risk stratum clinical profile: Baseline UPDRS 14.8$\pm$2.1, 94\% RBD+, 87\% hyposmic, DaTscan SBR 1.58$\pm$0.24, 28\% GBA+. Contrasted with low-risk (UPDRS 7.2, 38\% RBD+, SBR 2.42).
}
\label{fig:stratification}
\end{figure*}
